const fs = require('fs');
let html = fs.readFileSync('docs/Admin/vault_admin.html', 'utf8');
const search = `} else {
      document.getElementById("loginErr").textContent = e.name === 'AbortError'
        ? "Connection timed out — please try again"
        : "Incorrect PIN — try again";
    }`;
const replace = `} else {
      let errMsg = "Incorrect PIN — try again";
      if (e.name === 'AbortError') errMsg = "Connection timed out — please try again";
      else if (e.isMenuError && !e.isPinError) errMsg = "Google Server Busy — please retry";
      else if (e.message && e.message.includes('Unexpected')) errMsg = "Google Server Error — please retry";
      
      document.getElementById("loginErr").textContent = errMsg;
    }`;
// replace using regex to ignore line endings
const searchRegex = /\} else \{\s+document\.getElementById\("loginErr"\)\.textContent = e\.name === 'AbortError'\s+\? "Connection timed out — please try again"\s+: "Incorrect PIN — try again";\s+\}/;
html = html.replace(searchRegex, replace);
fs.writeFileSync('docs/Admin/vault_admin.html', html);
